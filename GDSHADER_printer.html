render_mode cull_disabled;

uniform vec3 shallow_color : source_color = vec3(0.3, 0.7, 0.9);
uniform vec3 deep_color : source_color = vec3(0.05, 0.2, 0.4);
uniform float wave_speed : hint_range(0.0, 4.0) = 1.0;
uniform float wave_height : hint_range(0.0, 1.0) = 0.15;

const float TWO_PI = 6.2831853;

float wave(vec2 p, float t) {
    return sin(p.x * 3.0 + t) * cos(p.y * 2.0 + t * 0.7);
}

void vertex() {
    float t = TIME * wave_speed;
    VERTEX.y += wave(VERTEX.xz, t) * wave_height;
}

void fragment() {
    float crest = clamp(wave(UV * TWO_PI, TIME * wave_speed) * 0.5 + 0.5, 0.0, 1.0);
    ALBEDO = mix(deep_color, shallow_color, crest);
    ALPHA = 0.8;
}
