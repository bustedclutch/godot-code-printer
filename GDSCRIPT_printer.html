@onready var adventure_log = $Control/AdventureLog
@onready var command_input = $Control/CommandInput

signal item_taken(item_id: String)
signal vault_reached

var current_room: String = "entry_hall"
var inventory: Array = []
var has_key: bool = false

const DIRECTION_ALIASES : Dictionary = {
	"n" : "north",
	"s" : "south",
	"w" : "west",
	"e" : "east"
}

# Room list. Name, description, exits-->room key, optional item.
var rooms := {
	"entry_hall" : {
		"name" : "Crumbling Entry Hall",
		"desc" : "Torchlight flickers across mossy stone. A heavy door looms to the north.",
		"exits" : {"north" : "great_hall"}
	},
	"great_hall" : {
		"name" : "Great Hall",
		"desc" : "Tattered banners hang from the rafters of a vast, echoing chamber.",
		"exits" : {"south" : "entry_hall", "east" : "armory", "west" : "library", "north" : "throne_room"}
	},
	"armory" : {
		"name" : "Dusty Armory",
		"desc" : "Rusted weapons line the walls. A small iron key glints on a workbench.",
		"exits" : {"west" : "great_hall"},
		"item" : "iron_key"
	},
	"library" : {
		"name" : "Forgotten Library",
		"desc" : "Stacks of decaying tomes lean precariously. A scroll lies open on a podium.",
		"exits" : {"east" : "great_hall"},
		"item" : "scroll"
	},
	"throne_room" : {
		"name" : "Shattered Throne Room",
		"desc" : "A cracked throne sits before a sealed gate. Only a key will open it.",
		"exits" : {"south" : "great_hall", "north" : "treasure_vault"}
	},
	"treasure_vault" : {
		"name" : "Treasure Vault",
		"desc" : "Mounds of gold glimmer in the lantern light. You've found it!",
		"exits" : {"south" : "throne_room"}
	}
}

func _ready() -> void:
	_describe_room()
	command_input.grab_focus()
	item_taken.connect(_on_item_taken)
	vault_reached.connect(_on_vault_reached)

func _input(event: InputEvent) -> void:
	if event is InputEventKey and event.pressed and not event.echo:
		if event.keycode == KEY_ENTER and command_input.has_focus():
			_on_command(command_input.text)
			get_viewport().set_input_as_handled()

func _on_command(text: String) -> void:
	command_input.clear()
	var cmd : String = text.strip_edges().to_lower()
	if cmd == "":
		return
	_log("> " + cmd)

	##Three options for taking in the direction command: inline dict, match,
	##and a named dict constant. Type all of them since clearly you'll need
	##all three at various points! This is becoming good dictionary practice.
	#
	#if cmd in ["n", "s", "e", "w"]:cmd={"n":"north","s":"south","e":"east","w":"west"}[cmd]
	#
	#var aliases : Dictionary = {"n":"north","s":"south","w":"west","e":"east"}
	#cmd = aliases[cmd]
	#
	#match cmd:
		#"n" : cmd = "north"
		#"s" : cmd = "south"
		#"e" : cmd = "east"
		#"w" : cmd = "west"

	if cmd in DIRECTION_ALIASES:
		cmd = DIRECTION_ALIASES[cmd]

	if cmd == "take":
		_try_take()
	elif cmd in rooms[current_room].exits:
		var next_room : String = rooms[current_room].exits[cmd]
		if next_room == "treasure_vault" and not has_key:
			_log("The gate is sealed. You need a key.")
		else:
			current_room = next_room
			_describe_room()
			if current_room == "treasure_vault":
				vault_reached.emit()
	else:
		_log("You can't go that way.")

func _try_take() -> void:
	var room : Dictionary = rooms[current_room]
	if "item" in room:
		var item : String = room.item
		inventory.append(item)
		room.erase("item")
		if item == "iron_key":
			has_key = true
		item_taken.emit(item)
		_log("You picked up the " + item.replace("_", " ") + ".")
	else:
		_log("There is nothing here to take.")

func _describe_room() -> void:
	var room : Dictionary = rooms[current_room]
	_log("[b]%s[/b]" % room.name.capitalize())
	_log(room.desc)
	_log("Exits: " + ", ".join(room.exits.keys()))

func _on_item_taken(item_id: String) -> void:
	print("Picked up: " + item_id)

func _on_vault_reached() -> void:
	_log("You've reached the vault! Adventure complete.")

func _log(text: String) -> void:
	adventure_log.append_text(text + "\n")
